
dyson_gun_firing = {
	from.owner = {
		export_resource_stockpile_to_variable = {
			resource = energy
			variable = dyson_gun_charge
		}
		set_resource = {
			resource = energy
			value = 0
		}
	}
	if = {
		limit = {
			from = {
				is_megastructure_type = dyson_gun_1
			}
		}
		from.owner = {
			divide_variable = {
				which = dyson_gun_charge
				value = 100
			}
		}
		every_fleet_in_system = {
			limit = { NOR = {
				any_owned_ship = { exists = colony }
				AND = {
					exists = owner
					owner = { OR = {
						ag_is_psionic_country = yes
						ag_is_ancient_country = yes
						is_invisable_faction = yes
						is_country_type = ag_ancient_primitive_crisis
						has_country_flag = ag_invincible_on_system_destroyed_country
					} }
				}
			} }
			ag_dyson_gun_damage = yes
		}
		every_system_planet = {
			limit = {
				is_colony = yes
			}
			add_planet_devastation = 10
		}
	}
	if = {
		limit = {
			from = {
				is_megastructure_type = dyson_gun_2
			}
		}
		from.owner = {
			divide_variable = {
				which = dyson_gun_charge
				value = 10
			}
		}
		every_fleet_in_system = {
			limit = { NOR = {
				any_owned_ship = { exists = colony }
				AND = {
					exists = owner
					owner = { OR = {
						ag_is_psionic_country = yes
						ag_is_ancient_country = yes
						is_invisable_faction = yes
						is_country_type = ag_ancient_primitive_crisis
						has_country_flag = ag_invincible_on_system_destroyed_country
					} }
				}
			} }
			kuat_dyson_gun_damage = yes
		}
		every_system_planet = {
			limit = {
				is_colony = yes
			}
			add_planet_devastation = 25
		}
	}
	if = {
		limit = {
			from = {
				is_megastructure_type = dyson_gun_3
			}
		}
		from.owner = {
			divide_variable = {
				which = dyson_gun_charge
				value = 1
			}
		}

		every_fleet_in_system = {
			limit = { NOR = {
				any_owned_ship = { exists = colony }
				AND = {
					exists = owner
					owner = { OR = {
						ag_is_psionic_country = yes
						ag_is_ancient_country = yes
						is_invisable_faction = yes
						is_country_type = ag_ancient_primitive_crisis
						has_country_flag = ag_invincible_on_system_destroyed_country
					} }
				}
			} }
			kuat_dyson_gun_damage = yes
		}
		every_system_planet = {
			limit = {
				is_colony = yes
			}
			add_planet_devastation = 50
		}
		every_system_planet = {
			limit = {
				from.owner = {
					check_variable = {
						which = dyson_gun_charge
						value > 50000
					}
				}
			}
			increase_planet_temperature_without_habitables = yes
		}
		if = {
			limit = {
				OR = {
					is_active_resolution = "resolution_ecology_environmental_control_board"
					is_active_resolution = "resolution_ecology_paradise_initiative"
				}
				from.owner = {
					check_variable = {
						which = dyson_gun_charge
						value > 50000
					}
				}
			}
			from.owner = {
				set_timed_country_flag = { flag = resolution_breached_terraformed days = 3600 }
			}
		}
	}
	if = {
		limit = {
			space_owner? = {
				NOT = {
					is_same_value = from.owner
				}
			}
		}
		space_owner = {
			create_message = {
				type = MESSAGE_DYSON_SHOT
				localization =  MESSAGE_DYSON_SHOT_DESC
				days = @toast_message_days
			}
			add_opinion_modifier = {
				who = from.owner
				modifier = opinion_dyson_gun_target
			}
		}
	}
	if = {
		limit = {
			any_system_planet = { is_colony = yes }
			OR = {
				is_active_resolution = "resolution_rulesofwar_independent_tribunals"
				is_active_resolution = "resolution_rulesofwar_last_resort_doctrine"
				is_active_resolution = "resolution_rulesofwar_demobilization_initiative"
			}
		}
		from.owner = {
			set_timed_country_flag = { flag = resolution_breached_fired_dyson_gun days = 3600 }
		}
	}
	if = {
		limit = {
			any_system_planet = { is_homeworld = yes }
		}
		from.owner = {
			set_country_flag = fired_dyson_gun_at_homeworld
		}
	}
	# Destroy impact ambient objects after a short delay
	system_event = {
		id = nomads.8005
		days = 5
	}
}
kuat_dyson_gun_damage = {
	optimize_memory
	every_ship_in_system = {
		limit = {
			check_modifier_value = { modifier = ag_ship_aoe_damage_reduction value < 1.0 }
			check_modifier_value = { modifier = touhou_ship_aoe_damage_reduction_mult value < 1.0 }
		}
		kuat_battle_cretae_aoe_particle_effect = yes
		root = { from = { prevprev = { set_timed_ship_flag = { flag = kuat_aoe_damaged_@prev days = @kuat_global_dynamic_hitpoints } } } }
		set_variable = { which = kuat_aoe_total_damage_value value = root.kuat_aoe_total_damage_value_bak }
		multiply_variable = { which = kuat_aoe_total_damage_value value = value:kuat_modifier_mult_subtract|kuat_modifier|ship_damage_reduction_mult|kuat_min|0| }
		multiply_variable = { which = kuat_aoe_total_damage_value value = value:kuat_modifier_mult_subtract|kuat_modifier|touhou_ship_aoe_damage_reduction_mult|kuat_min|0| }
        subtract_variable = { which = kuat_aoe_total_damage_value value = modifier:ship_damage_reduction_add }
		subtract_variable = { which = kuat_aoe_total_damage_value value = modifier:touhou_ship_aoe_damage_reduction_add }
		multiply_variable = { which = kuat_aoe_total_damage_value value = value:kuat_modifier_mult_subtract|kuat_modifier|ag_ship_aoe_damage_reduction|kuat_min|0| }
		if = {
			limit = {
				check_variable = { which = kuat_aoe_total_damage_value value > 0 }
				has_shield_hp > 0
			}
			set_variable = { which = kuat_aoe_shield_damage_value value = kuat_aoe_total_damage_value }
			multiply_variable = {
				which = kuat_aoe_shield_damage_value
				value = from.value:kuat_modifier_mult_add|kuat_modifier|ship_shield_damage_mult|
			}
			if = {
				limit = { has_shield_hp >= kuat_aoe_shield_damage_value }
				reduce_shield = kuat_aoe_shield_damage_value
			}
			else = {
				set_variable = { which = kuat_aoe_shield_damage_value value = trigger:has_shield_hp }
				divide_variable = {
					which = kuat_aoe_shield_damage_value
					value = from.value:kuat_modifier_mult_add|kuat_modifier|ship_shield_damage_mult|
				}
				reduce_shield = trigger:has_shield_hp
			}
			subtract_variable = { which = kuat_aoe_total_damage_value value = kuat_aoe_shield_damage_value }
			clear_variable = kuat_aoe_shield_damage_value
		}
		if = {
			limit = {
				check_variable = { which = kuat_aoe_total_damage_value value > 0 }
				has_armor_percentage > 0
			}
			set_variable = { which = kuat_currently_armor_hp value = value:kuat_has_armor_hp }
			set_variable = { which = kuat_aoe_armor_damage_value value = kuat_aoe_total_damage_value }
			multiply_variable = {
				which = kuat_aoe_armor_damage_value
				value = from.value:kuat_modifier_mult_add|kuat_modifier|ship_armor_damage_mult|
			}
			if = {
				limit = { check_variable = { which = kuat_currently_armor_hp value >= kuat_aoe_armor_damage_value } }
				multiply_variable = { which = kuat_aoe_armor_damage_value value = -1 }
				repair_armor_amount = kuat_aoe_armor_damage_value
			}
			else = {
				set_variable = { which = kuat_aoe_armor_damage_value value = kuat_currently_armor_hp }
				multiply_variable = { which = kuat_aoe_armor_damage_value value = -1 }
				repair_armor_amount = kuat_aoe_armor_damage_value
				set_variable = { which = kuat_aoe_armor_damage_value value = kuat_currently_armor_hp }
				divide_variable = {
					which = kuat_aoe_armor_damage_value
					value = from.value:kuat_modifier_mult_add|kuat_modifier|ship_armor_damage_mult|
				}
				multiply_variable = { which = kuat_aoe_armor_damage_value value = -1 }
			}
			change_variable = { which = kuat_aoe_total_damage_value value = kuat_aoe_armor_damage_value }
			clear_variable = kuat_currently_armor_hp
			clear_variable = kuat_aoe_armor_damage_value
		}
		if = {
			limit = { check_variable = { which = kuat_aoe_total_damage_value value > 0 } }
			multiply_variable = { which = kuat_aoe_total_damage_value value = from.value:kuat_modifier_mult_add|kuat_modifier|ship_hull_damage_mult| }
			change_variable = { which = kuat_aoe_total_damage_value value = modifier:kuat_weapon_aoe_ship_hull_damage }
			add_modifier = { modifier = kuat_ship_weapon_on_hit_aoe_modifier multiplier = kuat_aoe_total_damage_value days = 2 }
		}
		else = { clear_variable = kuat_aoe_total_damage_value }
	}
}
