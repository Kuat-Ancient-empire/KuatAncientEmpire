kuat_galatic_empire_ge_invasion_create_secondary_portal_system = {
	optimize_memory
	random_system = {
		limit = { NOR = {
			has_star_flag = kuat_galatic_empire_e_area_system
			has_star_flag = kuat_galatic_empire_portal_system
		} }
		spawn_system = {
			min_distance >= 20
			max_distance <= 50
			initializer = "kuat_galatic_empire_portal_system_init"
			hyperlane = yes
		}
		last_created_system = {
			save_global_event_target_as = kuat_galatic_empire_portal_system
			if = {
				limit = { event_target:global_event_country = { check_variable = { which = kuat_galatic_empire_ge_invasion_num_portals value = 1 } } }
				set_star_flag = kuat_galatic_empire_portal_system_1
			}
			else_if = {
				limit = { event_target:global_event_country = { check_variable = { which = kuat_galatic_empire_ge_invasion_num_portals value = 2 } } }
				set_star_flag = kuat_galatic_empire_portal_system_2
			}
			else_if = {
				limit = { event_target:global_event_country = { check_variable = { which = kuat_galatic_empire_ge_invasion_num_portals value = 3 } } }
				set_star_flag = kuat_galatic_empire_portal_system_3
			}
			else_if = {
				limit = { event_target:global_event_country = { check_variable = { which = kuat_galatic_empire_ge_invasion_num_portals value = 4 } } }
				set_star_flag = kuat_galatic_empire_portal_system_4
			}
			else_if = {
				limit = { event_target:global_event_country = { check_variable = { which = kuat_galatic_empire_ge_invasion_num_portals value = 5 } } }
				set_star_flag = kuat_galatic_empire_portal_system_5
			}
			else_if = {
				limit = { always = yes }
				set_star_flag = kuat_galatic_empire_portal_system_6
			}
		}
	}
	event_target:global_event_country = { change_variable = { which = kuat_galatic_empire_ge_invasion_num_portals value = 1 } }
}

kuat_galatic_empire_ge_invasion_primary_portal_spawn = {
	optimize_memory
	solar_system = {
		spawn_megastructure = {
			type = "kuat_galatic_empire_ge_invasion_crisis_portal"
			owner = event_target:Kuat_Galatic_Empire_invasion_country
			orbit_angle = 225
			orbit_distance = 200
			planet = star
		}
		set_star_flag = cannot_be_hit_by_aow_starfall_spear
		set_star_flag = kuat_cannot_be_hit_by_starkillerbase
	}
	set_planet_size = -1
	set_planet_entity = { entity = starforge_avoid_entity }
	create_fleet = {
		name = "NAME_kuat_galatic_empire_ge_portal_fleet"
		settings = { spawn_debris = no can_disband = no is_boss = yes }
		effect = {
			set_owner = event_target:Kuat_Galatic_Empire_invasion_country
			create_ship = { 
				name = "NAME_kuat_galatic_empire_ge_portal" 
				design = "NAME_EXE_STARBASE_MEGASTRUCTURE" prefix = no 
				effect = { 
					set_variable = { which = STARBASE_HEALTH_VAR value = 0 }
					kuat_battle_disable_ship_at_health = { 
						FLAG = enable_disable_starbase 
						Disable_health = 0.01 
						WHICH = STARBASE_HEALTH_VAR 
						HEALTH = value:kuat_dynamic_exist_life_times 
					}
					set_ship_flag = kuat_galatic_empire_ge_primary_portal 
				}
			}
			set_location = { target = prev distance = -1 angle = 0 }
			set_fleet_flag = kuat_galatic_empire_ge_primary_portal
			set_fleet_flag = kuat_empire_starbase_hull_control
			set_event_locked = yes
			fleet_event = { id = kuat_galatic_empire_ge_invasion.262 days = 60 } 
		}
	}
	create_ambient_object = { type = "contingency_1" location = this }
	last_created_ambient_object = {
		set_ambient_object_flag = space_storm_object
		set_location = { target = prev distance = 0 angle = random }
	}
}
