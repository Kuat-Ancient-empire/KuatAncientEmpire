@kuat_flagship_proection_mult = 0.1
kuat_battle_eternal_increase_executor_power = {
	optimize_memory
	if = {
		limit = { NOT = { has_ship_flag = kuat_has_eternal_increase_executor_power_temple } }
		if = {
			limit = { kuat_is_disable_return_ship = yes owner = { kuat_is_player_country = yes } }
			set_variable = { which = KUAT_SHIP_HEALTH_VAR value = 0 }
			kuat_battle_disable_ship_at_health = { 
				FLAG = enable_kuat_ship_inital Disable_health = 0.01 
				WHICH = KUAT_SHIP_HEALTH_VAR HEALTH = value:kuat_dynamic_exist_life_times 
			}
		}
		add_modifier = { 
			modifier = kuat_harding_hp 
			multiplier = value:kuat_dynamic_mutiply_factor|BASE|trigger:has_max_hp|kuat_ship_damage_buff|modifier:ship_hull_mult|MULT|0.00002| 
		}
		add_modifier = { 
			modifier = kuat_regen_hp 
			multiplier = value:kuat_dynamic_mutiply_factor|BASE|trigger:has_max_hp|kuat_ship_damage_buff|modifier:ship_hull_mult|MULT|0.001| 
		}
		set_ship_flag = kuat_has_eternal_increase_executor_power_temple
	}
}

kuat_battle_eternal_increase_executor_hp_power = {
	optimize_memory
	if = {
		limit = { NOT = { has_ship_flag = kuat_has_eternal_increase_executor_power_hp_temple } }
		kuat_battle_clear_ship_modifier = { modifier = ship_hull_mult DAYS = $DAYS$ subtract = @kuat_flagship_proection_mult }
		kuat_battle_clear_ship_modifier = { modifier = shipclass_military_hull_mult DAYS = $DAYS$ subtract = @kuat_flagship_proection_mult }
		kuat_battle_clear_ship_modifier = { modifier = ship_hull_regen_add_perc DAYS = $DAYS$ subtract = @kuat_flagship_proection_mult }

		if = {
			limit = { has_hp < 100000000 }
			add_modifier = { modifier = kt_increase_hp days = $DAYS$ multiplier = value:kuat_eternal_increase_hp_value }
		}

		set_timed_ship_flag = { flag = kuat_has_eternal_increase_executor_power_hp_temple days = $DAYS$ }
	}
}

# tip:script scope available in penetration modifier, other will be error.
# parameter： modifier.
# OUTPUT_VALUE: target value of modifier.
# TYPE: value class type, for example change\subtract.
# SUPPORT: + OR -  
kuat_battle_set_eternal_fleet_standard_modifier = {
	optimize_memory
	if = { 
		limit = { 
			NOR = { 
				has_ship_flag = $PARAMATER$_ship_cooldown 
				has_ship_flag = kuat_is_modifiering
			}
			check_modifier_value = { modifier = $PARAMATER$ value $RELATION$ $OUTPUT_VALUE$ }
		}
		set_variable = { which = $PARAMATER$_value value = modifier:$PARAMATER$ }
		subtract_variable = { which = $PARAMATER$_value value = $OUTPUT_VALUE$ }
		add_modifier = { modifier = basic_$PARAMATER$ multiplier = $PARAMATER$_value }
		if = { limit = { is_variable_set = $PARAMATER$_value } clear_variable = $PARAMATER$_value }
		set_timed_ship_flag = { flag = $PARAMATER$_ship_cooldown days = 1 }
		if = { 
			limit = { NOT = { has_ship_flag = kuat_fleet_dynamic_modifier_times } } 
			set_timed_ship_flag = { flag = kuat_fleet_dynamic_modifier_times days = 1 }
		}
	} 
	else_if = { 
		limit = { 
			has_ship_flag = kuat_is_modifiering
			NOT = { has_ship_flag = $PARAMATER$_ship_cooldown }
			check_modifier_value = { modifier = $PARAMATER$ value $RELATION$ $OUTPUT_VALUE$ }
		}
		set_variable = { which = change_$PARAMATER$_value value = modifier:$PARAMATER$ }
		subtract_variable = { which = change_$PARAMATER$_value value = $OUTPUT_VALUE$ }
		change_variable = { which = debug_change_$PARAMATER$_value value = change_$PARAMATER$_value }
		add_modifier = { modifier = change_basic_$PARAMATER$ multiplier = debug_change_$PARAMATER$_value }
		if = { limit = { is_variable_set = change_$PARAMATER$_value } clear_variable = change_$PARAMATER$_value }
		set_timed_ship_flag = { flag = $PARAMATER$_ship_cooldown days = 1 }
		if = { 
			limit = { NOT = { has_ship_flag = kuat_fleet_dynamic_modifier_times } } 
			set_timed_ship_flag = { flag = kuat_fleet_dynamic_modifier_times days = 1 }
		}
	}
}

			NOR = { 
				has_ship_flag = repeatable_weapon_type_$TYPE$_weapon_fire_rate_mult_ship_cooldown 
				has_ship_flag = kuat_is_modifiering
			} 
			NOT = { check_modifier_value = { modifier = weapon_type_$TYPE$_weapon_fire_rate_mult value = $OUTPUT_VALUE$ } }
		}

		set_variable = { 
			which = weapon_type_$TYPE$_weapon_fire_rate_mult_value 
			value = modifier:weapon_type_$TYPE$_weapon_fire_rate_mult 
		}
		subtract_variable = { which = weapon_type_$TYPE$_weapon_fire_rate_mult_value value = $OUTPUT_VALUE$ }	
		add_modifier = { 
			modifier = basic_weapon_type_$TYPE$_weapon_fire_rate_mult

		}	
		if = {
			limit = { is_variable_set = weapon_type_$TYPE$_weapon_fire_rate_mult_value }		
			clear_variable = weapon_type_$TYPE$_weapon_fire_rate_mult_value 
		}
		set_timed_ship_flag = { flag = repeatable_weapon_type_$TYPE$_weapon_fire_rate_mult_ship_cooldown days = 1 }
	} 
	else_if = { 
		limit = { 
			has_ship_flag = kuat_is_modifiering
			NOT = { has_ship_flag = repeatable_weapon_type_$TYPE$_weapon_fire_rate_mult_ship_cooldown } 
			NOT = { check_modifier_value = { modifier = weapon_type_$TYPE$_weapon_fire_rate_mult value = $OUTPUT_VALUE$ } }
		}
		set_variable = { 
			which = change_weapon_type_$TYPE$_weapon_fire_rate_mult_value 
			value = modifier:weapon_type_$TYPE$_weapon_fire_rate_mult 
		}
		subtract_variable = { which = change_weapon_type_$TYPE$_weapon_fire_rate_mult_value value = $OUTPUT_VALUE$ }
		change_variable = { 
			which = debug_change_weapon_type_$TYPE$_weapon_fire_rate_mult_value 
			value = change_weapon_type_$TYPE$_weapon_fire_rate_mult_value 
		}
		add_modifier = { 
			modifier = change_basic_weapon_type_$TYPE$_weapon_fire_rate_mult 
			multiplier = debug_change_weapon_type_$TYPE$_weapon_fire_rate_mult_value 
		}
		if = {
			limit = { is_variable_set = change_weapon_type_$TYPE$_weapon_fire_rate_mult_value }	
			clear_variable = change_weapon_type_$TYPE$_weapon_fire_rate_mult_value 
		}
		set_timed_ship_flag = { flag = repeatable_weapon_type_$TYPE$_weapon_fire_rate_mult_ship_cooldown days = 1 }
	}
}
