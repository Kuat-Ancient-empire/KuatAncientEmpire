# scope: ship
@kuat_flagship_proection_mult = 0.1
kuat_battle_eternal_increase_executor_power = {
	optimize_memory
	if = {
		limit = { NOT = { has_ship_flag = kuat_has_eternal_increase_executor_power_temple } }
		if = {
			limit = { 
				kuat_is_disable_return_ship = yes 
				owner? = { is_invisable_faction = no }
				NOT = { has_global_flag = kuat_set_forcetrigger_order_enable_crisis_eve_system }
			}
			set_variable = { which = KUAT_SHIP_HEALTH_VAR value = 0 }
			kuat_battle_disable_ship_at_health = { 
				FLAG = enable_kuat_ship_inital Disable_health = 0.01 
				WHICH = KUAT_SHIP_HEALTH_VAR HEALTH = value:kuat_dynamic_exist_life_times 
			}
			kuat_battle_eternal_initial_started_defence_value = { MULT = 0.001 }
		}
		else = { kuat_battle_eternal_initial_started_defence_value = { MULT = 0.0005 } }
		if = {
			limit = { 
				owner? = { is_invisable_faction = no }
				NOT = { has_global_flag = kuat_set_forcetrigger_order_enable_crisis_eve_system }
				check_modifier_value = { modifier = ship_damage_reduction_mult value < value:kuat_dynamic_ship_damage_reduce_limit } 
			}
			kuat_battle_clear_ship_modifier = { modifier = ship_damage_reduction_mult subtract = value:kuat_dynamic_ship_damage_reduce_limit DAYS = -1 }
		}
		set_ship_flag = kuat_has_eternal_increase_executor_power_temple
	}
}

	optimize_memory

		multiplier = value:kuat_dynamic_mutiply_factor|MULT|$MULT$|kuat_mult_modifier|ship_hull_mult|kuat_add_modifier|ship_hull_add|

	add_modifier = {

kuat_battle_setup_dynamic_invinciable_protection_pragram = {
	optimize_memory
	if = {
		limit = { is_in_combat = yes }
		every_owned_ship = {
			limit = { check_modifier_value = { 
				modifier = kuat_damage_control_modifier value > 0 
			} }
			$KEY$ = { frendly = yes }
			if = {
				limit = { NOT = { 
					has_ship_flag = kuat_is_modifiering  
				} }
				set_timed_ship_flag = { 
					flag = kuat_is_modifiering 
					days = @kuat_global_dynamic_delay_times 
				}
			}
			set_confused = no remove_terrified = yes
		}
		fleet_event = { id = $ID$ days = @kuat_global_dynamic_hitpoints }
	}
	else = { kuat_battle_clear_battleship_modifier_fixed_ship = yes }
}

kuat_battle_set_kuat_special_damage_control_standard_modifier = {
	optimize_memory
	if = {
		limit = { 
			check_modifier_value = { modifier = $PARAMATER$ value $RELATION$ $OUTPUT_VALUE$ } 
			fleet? = { OR = { 
				has_fleet_flag = has_none_kuat_flagship_system 
				NOT = { has_fleet_flag = has_super_kuat_flagship_system }
			} }
		}
		set_variable = { which = $PARAMATER$_value value = modifier:$PARAMATER$ }
		subtract_variable = { which = $PARAMATER$_value value = $OUTPUT_VALUE$ }
		change_variable = { which = basic_$PARAMATER$_value value = $PARAMATER$_value }
		add_modifier = { modifier = basic_$PARAMATER$ multiplier = basic_$PARAMATER$_value days = $DAYS$ } 
		clear_variable = $PARAMATER$_value
	}
}

kuat_battle_flagship_intial = {
	optimize_memory
	if = {
		limit = { OR = {
			has_global_flag = easy_exe_invasion
			has_global_flag = normal_exe_invasion
			has_global_flag = kuat_set_forcetrigger_order_enable_crisis_eve_system
			root.from? = { is_invisable_faction = yes }
		} }
		set_fleet_flag = has_total_kuat_flagship_system
		set_fleet_flag = has_super_kuat_flagship_system
	}
	else_if = {
		limit = { OR = {
			has_global_flag = ban_exe_invasion
			has_global_flag = difficult_exe_invasion
		} }
		if = {
			limit = { OR = {
				kuat_exist_fleet_leader = { TRAIT = leader_trait_commander_SOP_eddy_3 }
				kuat_exist_fleet_leader = { TRAIT = leader_trait_eternal_learned_leader_legend }
			} }
			set_fleet_flag = has_total_kuat_flagship_system
			set_fleet_flag = has_super_kuat_flagship_system
		}
		else_if = {
			limit = { kuat_ship_is_mega_starbase = yes }
			set_fleet_flag = has_total_kuat_flagship_system
			set_fleet_flag = has_super_kuat_flagship_system
		}
		else_if = {
			limit = { kuat_fleet_anyone_has_validable_super_flagship_system = yes }
			set_fleet_flag = has_total_kuat_flagship_system
			set_fleet_flag = has_super_kuat_flagship_system
		}
		else_if = {
			limit = { 
				is_within_borders_of = owner
				kuat_fleet_anyone_has_validable_standard_flagship_system = yes
				kuat_fleet_anyone_has_validable_sub_flagship_system = yes
			}
			set_fleet_flag = has_super_kuat_flagship_system
			set_fleet_flag = has_total_kuat_flagship_system
		}
		else_if = {
			limit = { 
				kuat_fleet_anyone_has_validable_standard_flagship_system = yes
				kuat_fleet_anyone_has_validable_sub_flagship_system = yes
			}
			set_fleet_flag = has_total_kuat_flagship_system
		}
		else_if = {
			limit = { 
				is_within_borders_of = owner
				kuat_fleet_anyone_has_validable_standard_flagship_system = yes
			}
			set_fleet_flag = has_total_kuat_flagship_system
		}
		else_if = {
			limit = {
				NOT = { kuat_fleet_anyone_has_validable_standard_flagship_system = yes }
			}
			set_fleet_flag = has_none_kuat_flagship_system
		}
	}
}
