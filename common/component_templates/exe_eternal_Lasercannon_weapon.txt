# 图标：GFX_ship_part_kuat_$kuat_index$_$kuat_weapon_type$_$kuat_weapon_name$_$kuat_size$_$kuat_component_type$
# 科技：tech_kuat_$kuat_index$_$kuat_weapon_type$_$kuat_weapon_name$
# 由于组件不能一个inline注册多个，所以需要写多份并在kuat_size和kuat_component_type上做差分，其他属性会自动计算。
weapon_component_template = {
    inline_script = {
        script = components/kuat_template_common_weapon
		kuat_icon = "GFX_ship_part_etim_laser_blue_01"					# 图标
        kuat_index = ETIM													# 名称字符串1
        kuat_weapon_type = LASER_CANNON										# 名称字符串2
        kuat_weapon_name = ETM_MK1											# 名称字符串3
        kuat_size = small													# 武器规格，会自动放进名称里(titanic,extra_large,large,medium,small,point_defence)
        kuat_component_type = standard_ship									# 舰船分类，会自动放进组件名称和component_set里(standard_ship代表普通舰船，bio_ship代表生物舰船，mutation代表动物朋友)
        kuat_damage_type = instant											# 伤害类型，包含(instant,missile和point_defence)
        kuat_prio_projectile = no
        kuat_mutation_weapon_fixup = no										# 动物朋友武器的x槽伤害修正，如果原型武器是x就写extra_large,否则写no,类似的取值有torpedo,missile,point_defence和strike_craft，至于为什么这么多差分，要问蠢驴。
        kuat_weapon_tag = "weapon_type_explosive"						# 默认带了难度修正所以tag有前后之分，以移除
        kuat_extra_tag = "s_slot"
        kuat_ai_tags = "weapon_role_anti_hull"
        kuat_projectile_gfx = "sws_laser_cannon_small_purple"
        kuat_standard_cost = "energy = 1.0"									# 1.0即按倍率乘算。
        kuat_rare_cost = "alloys = 1.0"
        kuat_standard_upkeep = "energy = 0.1"
        kuat_rare_upkeep = "alloys = 0.2"
        kuat_has_upgrade = yes																		# 有上级组件写yes
        kuat_upgrade_key = "ETIM_LASER_CANNON_ETM_MK2"												# 写上级组件的$kuat_index$_$kuat_weapon_type$_$kuat_weapon_name$
        kuat_ai_weight = "1"
        kuat_weight_modifier = "modifier = { factor = 5000 is_ai = no } modifier = { factor = 0 is_ai = yes }"
        kuat_extra_weapon_datas = "no"																# 是导弹写missile，是舰载机写strike_craft，否则写no。

        kuat_tech_overwrite = yes																	# 写yes强制覆盖科技的key
        kuat_tech = "tr_eternal_laser_mk1_cannon"

        kuat_component_set_overwrite = yes															# 写yes强制覆盖component_set的key
        kuat_component_set = "ETIM_ETM_MK1_LASER_CANNON"

        kuat_has_firing_arc = no																	# 写yes才有发射角度

		kuat_component_set_size_restriction = no
		kuat_shipsize = ""

		kuat_trigger = "always = yes"
    }
}
