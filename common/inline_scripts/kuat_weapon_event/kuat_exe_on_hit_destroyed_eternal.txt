on_hit = {
	custom_tooltip_with_params = {
		description = "kuat_weapon_erase_damage_reduce_on_hit_effect"
		description_parameters = {
			"var_1" = $kuat_weapon_stollen_value$
			"var_2" = $kuat_weapon_stollen_days$
		}
	}
	hidden_effect = {
		if = {
			limit = { check_modifier_value = { modifier = ship_damage_reduction_add value > 0 } }
			kuat_battle_steal_modifier_on_hit = {
				VARIABLE = eternal_ship_damage_reduce_reduce_temp
				MODIFIER = kuat_damage_steal_weapon_hit_modifier
				KUAT_MODIFIER = kuat_weapon_steal_ship_damage
				AMOUNT = $kuat_weapon_stollen_value$
				DAYS = $kuat_weapon_stollen_days$
			}
		}
		if = {
			limit = { 
				AND = { exists = root.from root.from = { is_invisable_faction = yes } }
				AND = { exists = owner owner = { is_invisable_faction = no } }
				NOT = { has_ship_flag = kuat_battle_ship_damage_reduction_temple }
			}
			if = {
				limit = { 
					check_modifier_value = { 
						modifier = ship_damage_reduction_mult 
						value > value:kuat_dynamic_ship_damage_reduce_limit 
					}
				}
				kuat_battle_steal_modifier_on_hit = {
					VARIABLE = kuat_battle_ship_damage_reduction_mult_temp
					KUAT_MODIFIER = kuat_weapon_steal_ship_damage_reduce_mult
					MODIFIER = kuat_ship_damage_reduction_mult_steal_weapon_hit_modifier
					AMOUNT = value:kuat_dynamic_ship_damage_reduce_limit 
					DAYS = $kuat_weapon_stollen_days$
				}
			}
			if = {
				limit = { 
					check_modifier_value = { 
						modifier = ship_damage_reduction_add
						value > value:kuat_dynamic_mutiply_factor_limit|MULT|0.001|kuat_add_modifier|ship_hull_add|
					} 
				}
				kuat_battle_steal_modifier_on_hit = {
					VARIABLE = kuat_battle_ship_damage_reduction_add_temp
					KUAT_MODIFIER = kuat_weapon_steal_ship_damage_reduce_add
					MODIFIER = kuat_ship_damage_reduction_add_steal_weapon_hit_modifier
					AMOUNT = value:kuat_dynamic_mutiply_factor_limit|MULT|0.001|kuat_add_modifier|ship_hull_add|
					DAYS = $kuat_weapon_stollen_days$
				}
			}
			else = { set_timed_ship_flag = { flag = kuat_battle_ship_damage_reduction_temple days = $kuat_weapon_stollen_days$ } }
		}
	}
}
