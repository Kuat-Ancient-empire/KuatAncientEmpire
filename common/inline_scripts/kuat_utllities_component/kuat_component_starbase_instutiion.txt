#释放补给风暴
scripted_action = {
	scope = self

	button_visible = {
		any_owned_ship = { kuat_ship_is_mega_starbase = yes }
	}

	button_clickable = {
		custom_tooltip = {
			NOT = { has_fleet_flag = kuat_mega_starbase_stormed_activated }
			any_owned_ship = { kuat_ship_is_mega_starbase = yes }
		}
	}

	inline_script = {
		script = "kuat_utllities_component/kuat_component_defines"
		CLASS = kuat_mega_starbase_stormed
	}

	icon = GFX_fleet_action_kuat_icon_supplies
	icon_selected = GFX_fleet_action_kuat_icon_supplies
	cooldown = 1

	shortcut = "SHIFT+S"
	slot = 1
}
