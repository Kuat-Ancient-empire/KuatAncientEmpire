scripted_action = {
	scope = self

	button_visible = {
		OR = {
			is_ship_class = shipclass_military
			is_ship_class = shipclass_military_special
			is_ship_class = shipclass_transport
		}
	}

	button_clickable = {
		custom_tooltip = {
			text = "kuat_fleet_attack_action_allow"
			exists = owner owner = { is_at_war = yes }
		}
		custom_tooltip = {
			text = "kuat_fleet_attack_defender_allow"
			NOT = { has_fleet_flag = fire_kuat_fleet_attack_action }
		}
	}
	wait = no
	inline_script = {
		script = "kuat_utllities_component/kuat_component_defines"
		CLASS = kuat_auto_dreadnought_fleet_action
	}
	icon = GFX_fleet_action_kuat_icon_fleet_auto_action
	icon_selected = GFX_fleet_action_kuat_icon_fleet_auto_action
	cooldown = 1
	shortcut = "SHIFT+A"
	slot = 1
}
