kuat_mega_starbase_stormed = {
	user_scope = fleet
	scope = self

	button_visible = {
		any_owned_ship = { kuat_ship_is_mega_starbase = yes }
	}

	button_clickable = {
		custom_tooltip = {
			NOT = { has_fleet_flag = kuat_mega_starbase_stormed_activated }
			any_owned_ship = { kuat_ship_is_mega_starbase = yes }
		}
	}

	inline_script = {
		script = "kuat_utllities_component/kuat_component_defines"
		CLASS = kuat_mega_starbase_stormed
	}
	wait = no
	icon = GFX_fleet_action_kuat_icon_supplies
	icon_selected = GFX_fleet_action_kuat_icon_supplies
	cooldown = 1

	shortcut = "SHIFT+S"
	slot = 1
}

kuat_mega_starbase_special_explosion = {
	user_scope = fleet
	scope = self
	button_visible = {
		is_ship_size = kuat_target_signal_ship
	}
	button_clickable = {
		custom_tooltip = { 
            fail_text = "executor_build.1008.check.flag"
            owner = { NOT = { has_country_flag = kuat_kuat_mega_starbase_special_explosion_temple } }
        }
		custom_tooltip = {
			fail_text = "kuat_requires_outside_border_star_killer"
			solar_system = { NOT = { is_inside_border = prev.owner } }
		}
		custom_tooltip = {
			fail_text = "kuat_requires_no_kuat_system" 
			solar_system = { NOT = { has_star_flag = kuat_cannot_be_hit_by_starkillerbase } }
		}
		custom_tooltip = {
			fail_text = "kuat_requires_a_star_star_killer" 
			solar_system = { count_fleet_in_system = {
				count = 0
				limit = { exists = owner owner = { is_first_galatic_empire_faction = yes } }
			} }
		}
	}
	wait = no
	inline_script = {
		script = "kuat_utllities_component/kuat_component_defines"
		CLASS = kuat_mega_starbase_special_explosion
	}
	icon = GFX_fleet_action_kuat_icon_explosion
	icon_selected = GFX_fleet_action_kuat_icon_explosion
	cooldown = 1
	cost = { sr_dark_matter = 1000000 }
	shortcut = "SHIFT+E"
	slot = 3
}

	user_scope = fleet
	scope = self

	button_visible = {
		OR = {
			is_ship_class = shipclass_military
			is_ship_class = shipclass_military_special
			is_ship_class = shipclass_transport
		}
	}

	button_clickable = {
		custom_tooltip = {
			text = "kuat_fleet_attack_action_allow"
			exists = owner owner = { is_at_war = yes }
		}
		custom_tooltip = {
			text = "kuat_fleet_attack_defender_allow"
			NOT = { has_fleet_flag = fire_kuat_fleet_attack_action }
		}
	}
	wait = no
	inline_script = {
		script = "kuat_utllities_component/kuat_component_defines"

	shortcut = "SHIFT+A"
	slot = 1
}
