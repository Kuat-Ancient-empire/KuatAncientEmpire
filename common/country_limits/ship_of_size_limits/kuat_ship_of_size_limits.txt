kuat_subflagships_limit = {
	ship_types = {
		X308_Balletor
		X308_Titan
		X308_Tyrant
	}
	base = @kuat_sub_flagship_limit_size # 1 * juggernaut's size_multiplier
	show = {
		is_scope_valid = yes
		OR = {
			has_existing_ship_design = X308_Balletor
			has_existing_ship_design = X308_Titan
			has_existing_ship_design = X308_Tyrant
		}
	}
}
