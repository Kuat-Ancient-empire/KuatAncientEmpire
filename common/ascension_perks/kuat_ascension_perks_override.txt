			fail_text = "requires_ascension_perks_3"

		}

	}

			set_ai_personality = hyperthermia_empire
			country_event = { id = infernals.1000 }
		}
		activate_crisis_progression = hyperthermia_path
	}

	ai_weight = { factor = 0 }
