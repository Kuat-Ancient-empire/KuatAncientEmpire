##GET KUAT SHIPYARD ALLOYS
kuat_shipyard_check_allow_resource = {
	optimize_memory
	resource_stockpile_compare = {
		resource = alloys
		value >= $second_alloys$
		multiplier = value:kuat_shipyard_value|FACTOR|1|
	}
}
kuat_shipyard_check_simple_allow_resource = {
	optimize_memory
	resource_stockpile_compare = {
		resource = alloys
		value >= $second_alloys$
		multiplier = value:kuat_shipyard_flagship_value|FACTOR|1|
	}
}

get_ship_alloys_allow = {
	optimize_memory
	if = {
		limit = { always = $HSD|no$ }
		kuat_shipyard_check_allow_resource = { second_alloys = @building_the_HSD_cost }
	}
	if = {
		limit = { always = $CLD|no$ } 
		kuat_shipyard_check_allow_resource = { second_alloys = @building_the_CLD_cost }
	}
	if = {
		limit = { always = $ISD|no$ } 
		kuat_shipyard_check_allow_resource = { second_alloys = @building_the_ISD_cost }
	}
	if = {
		limit = { always = $RSD|no$ }
		kuat_shipyard_check_allow_resource = { second_alloys = @building_the_RSD_cost }
	}
	if = {
		limit = { always = $XSD|no$ }
		kuat_shipyard_check_allow_resource = { second_alloys = @building_the_XSD_cost }
	}
	if = {
		limit = { always = $GSD|no$ }
		kuat_shipyard_check_allow_resource = { second_alloys = @building_the_GSD_cost }
	}
	if = {
		limit = { always = $DSDD|no$ }
		kuat_shipyard_check_simple_allow_resource = { second_alloys = @building_the_DSDD_cost }
	}
	if = {
		limit = { always = $ESDD|no$ }
		kuat_shipyard_check_simple_allow_resource = { second_alloys = @building_the_ESDD_cost }
	}
	if = {
		limit = { always = $VSDD|no$ }
		kuat_shipyard_check_simple_allow_resource = { second_alloys = @building_the_VSDD_cost }
	}
}

		check_variable = { which = GSD_$TYPE$ value > 0 }
		check_variable = { which = DSDD_$TYPE$ value > 0 }
		check_variable = { which = ESDD_$TYPE$ value > 0 }
		check_variable = { which = VSDD_$TYPE$ value > 0 }
	}
}
