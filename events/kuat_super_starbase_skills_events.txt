fleet_event = {
	id = kuat_starbase_skills.0
	hide_window = yes
	is_triggered_only = yes

	trigger = { 
		has_fleet_flag = kuat_mega_starbase_stormed_activated
		NOT = { has_fleet_flag = kuat_starbase_skills.0.temple } 
	}

	immediate = {
		remove_fleet_flag = kuat_mega_starbase_stormed_activated
		random_owned_ship = {
			limit = { kuat_ship_is_mega_starbase = yes }
			every_cosmic_storm = { 
				limit = { has_storm_flag = storm_kuat_support_nexus_storm_random_list_@prev } 
				kuat_destroy_cosmic_storm = yes 
			}	
		}
		set_timed_fleet_flag = { flag = kuat_starbase_skills.0.temple days = 1 }
	}
}
