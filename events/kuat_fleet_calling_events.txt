fleet_event = {
	id = kuat_fleet_calling.11
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = enable_ftl_ship_animation
		NOT = { has_global_flag = kuat_set_forcetrigger_order_enable_crisis_eve_system }
		any_owned_ship = { 
			kuat_is_animation_ftl_ship = yes 
		}
		from = { any_fleet_in_system = {
			exists = controller
			controller = { is_hostile = root.owner }
		} }
	}

	immediate = {  
		from = { 
			every_fleet_in_system = {
				limit = {
					OR = {
						exists = controller controller = { is_hostile = root.owner }
						is_same_value = root
					}
					NOT = { has_fleet_flag = kuat_fleet_calling.12_event_lock }
				}
				if = {
					limit = { exists = controller controller = { is_hostile = root.owner } }
					clear_orders = yes clear_fleet_actions = this
				}
				set_timed_fleet_flag = { flag = kuat_fleet_calling.12_event_lock_temple days = 12 }
				set_event_locked = yes set_fleet_flag = kuat_fleet_calling.12_event_lock
				fleet_event = { id = kuat_fleet_calling.12 days = 12 }
			}
		}
	}
}

	id = kuat_fleet_calling.13
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		any_owned_fleet = {
			NOT = { has_fleet_flag = kuat_fleet_calling.12_event_lock_temple }
			has_fleet_flag = kuat_fleet_calling.12_event_lock
		}
	}
	immediate = {
		every_owned_fleet = {
			limit = {
				NOT = { has_fleet_flag = kuat_fleet_calling.12_event_lock_temple }

			}
			set_event_locked = no
			remove_fleet_flag = kuat_fleet_calling.12_event_lock
		}
	}
}
