fleet_event = {
	id = kuat_fleet_calling.11
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = enable_ftl_ship_animation
		NOT = { has_global_flag = kuat_set_forcetrigger_order_enable_crisis_eve_system }
		any_owned_ship = { 
			kuat_is_animation_ftl_ship = yes 
		}
		from = { any_fleet_in_system = {
			exists = controller
			controller = { is_hostile = root.owner }
		} }
	}

	immediate = {  
		if = {
			limit = { NOT = { has_fleet_flag = kuat_fleet_calling_12_event_lock } }
			set_timed_fleet_flag = { flag = kuat_fleet_calling_12_event_lock_temple days = 12 }
			set_event_locked = yes set_fleet_flag = kuat_fleet_calling_12_event_lock
			fleet_event = { id = kuat_fleet_calling.12 days = 12 }
		}
		from = { 
			every_ship_in_system = {
				limit = { 
					exists = controller controller = { is_hostile = root.owner }
					NOT = { has_ship_flag = kuat_fleet_jump_drive_support_temple } 
					NOT = { check_modifier_value = { modifier = ship_speed_reduction value = 1 } }
				}
				set_variable = { which = kuat_fleet_jump_drive_support_temple value = modifier:ship_speed_reduction }
				subtract_variable = { which = kuat_fleet_jump_drive_support_temple value = 1 }
				add_modifier = { modifier = kuat_fleet_jump_drive_support multiplier = kuat_fleet_jump_drive_support_temple days = 12 }
				clear_variable = kuat_fleet_jump_drive_support_temple
				set_timed_ship_flag = { flag = kuat_fleet_jump_drive_support_temple days = 12 }
				fleet = { set_fleet_flag = kuat_fleet_jump_drive_support_@root }
			}	
		}
	}
}

country_event = {
	id = kuat_fleet_calling.13
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		any_owned_fleet = {
			NOT = { has_fleet_flag = kuat_fleet_calling_12_event_lock_temple }
			has_fleet_flag = kuat_fleet_calling_12_event_lock
		}
	}
	immediate = {
		every_owned_fleet = {
			limit = {
				NOT = { has_fleet_flag = kuat_fleet_calling_12_event_lock_temple }
				has_fleet_flag = kuat_fleet_calling_12_event_lock
			}
			every_galaxy_fleet = {
				limit = { 
					has_fleet_flag = kuat_fleet_jump_drive_support_@prev 
					NOT = { has_modifier = kuat_fleet_jump_drive_support }
				}
				add_modifier = { modifier = begin_modifier_updated_batch_modifier days = 1 }
				remove_fleet_flag = kuat_fleet_jump_drive_support_@prev
			}
			set_event_locked = no remove_fleet_flag = kuat_fleet_calling_12_event_lock
		}
	}
}
